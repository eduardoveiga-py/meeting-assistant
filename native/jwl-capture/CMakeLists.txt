cmake_minimum_required(VERSION 3.24)
project(MeetingAssistantJwlCapture LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OBS_HEADERS "" CACHE PATH "Pinned obs-studio source root")
if(NOT EXISTS "${OBS_HEADERS}/libobs/obs.h")
  message(FATAL_ERROR "Set OBS_HEADERS to obs-studio 31.0.3 source")
endif()
add_library(meeting-assistant-jwl-capture MODULE jwl_capture.cpp)
target_include_directories(meeting-assistant-jwl-capture PRIVATE "${OBS_HEADERS}/libobs" "${OBS_HEADERS}/libobs-winrt")
target_compile_definitions(meeting-assistant-jwl-capture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(meeting-assistant-jwl-capture PRIVATE /utf-8 /W4)
target_link_libraries(meeting-assistant-jwl-capture PRIVATE dwmapi)
include(CTest)
if(BUILD_TESTING)
  add_executable(jwl-binding-test binding_test.cpp)
  add_test(NAME jwl-binding COMMAND jwl-binding-test)
endif()
